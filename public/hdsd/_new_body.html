Hình Trụ 3D (Anchor C.top/bottom)

⏳
#cetz.canvas(length: 0.8cm, {
  import cetz.draw: *
  draw-cylinder(name: "C", radius: 2, height: 4, center: (0,0,0), stroke: blue + 1.5pt)
  content("C.top", [$O'$], anchor: "south")
  content("C.bottom", [$O$], anchor: "north")
})
