\begin{ex}%[1D3H2-5]%[TDMT12]%[Mui Doan]
	Giá trị $\lim\limits_{x\to -\infty}\dfrac{\sqrt{x^2-3^{2026}}-x}{x+1}$ 	bằng
	\choice
	{$0$}
	{$+\infty$}
	{$1$}
	{\True $-2$}
	\loigiai{
		Ta có
		\allowdisplaybreaks
		\begin{align*}
			L&=\lim\limits_{x\to -\infty}\dfrac{\sqrt{x^2-3^{2\,026}}-x}{x+1}\\
			&= \lim\limits_{x\to -\infty}\dfrac{x^2-3^{2\,026}-x^2}{(x+1)\left(\sqrt{x^2-3^{2\,026}}+x\right)}\\
			&= \lim\limits_{x\to -\infty}\dfrac{-3^{2\,026}}{(x+1)\left(\sqrt{x^2-3^{2\,026}}+x\right)}\\
			&=\lim\limits_{x\to -\infty}\dfrac{-3^{2\,026}}{x(x+1)\left(1-\sqrt{1-\dfrac{3^{2\,026}}{x^2}}\right)}\\
			&=\lim\limits_{x\to -\infty}\dfrac{-3^{2\,026}\left(1+\sqrt{1-\dfrac{3^{2\,026}}{x^2}}\right)}{x(x+1)\cdot\dfrac{3^{2\,026}}{x^2}}\\
			&= \lim\limits_{x\to -\infty}\dfrac{-x^2\left(1+\sqrt{1-\dfrac{3^{2\,026}}{x^2}}\right)}{x(x+1)}\\
			&=\lim\limits_{x\to -\infty}\dfrac{-x\left(1+\sqrt{1-\dfrac{3^{2\,026}}{x^2}}\right)}{x+1}=-2.
		\end{align*}
	}
\end{ex}

\begin{ex}%[2D4V1-3]%[TDMT12]%[Vũ Hồng Toàn]
	Cho hàm số $f(x)=x^2-2\cos x$. Gọi $F(x)$ là một nguyên hàm của hàm số $f(x)$. Biết rằng đồ thị hàm số $y=f'(x)$ và đồ thị hàm số $y=F(x)$ có cùng tâm đối xứng.
	\choiceTF[t]
	{\True $f'(x)=2x+2\sin x$}
	{\True Tâm đối xứng của đồ thị hàm số $y=f'(x)$ là gốc toạ độ $O$}
	{\True $F(x)=\dfrac{1}{3}x^3-2\sin x$}
	{\True Đồ thị hàm số $y=F(x)$ có hai điểm cực trị nằm về hai phía của trục tung}
	\loigiai{
		\begin{itemchoice}
			\itemch Ta có $f(x) = x^2 - 2\cos x\Rightarrow f'(x)= 2x + 2\sin x$.
			\itemch Hàm số $f'(x)= 2x + 2\sin x$ có tập xác định là $\mathscr{D}=\mathbb{R}$.\\
			Với mọi $x\in\mathscr{D}$. Ta có
			\[f'(-x) = 2(-x) + 2\sin(-x) = -2x - 2\sin x = -(2x + 2\sin x) = -f'(x).\]
			Do đó $f'(x)$ là hàm số lẻ nên đồ thị hàm số $y=f'(x)$ có tâm đối xứng là gốc tọa độ $O$.
			\itemch Vì $F(x)$ là một nguyên hàm của $f(x)$, nên
			\[F(x)=\displaystyle\int f(x)\,\mathrm{d}x=\displaystyle\int \left(x^2-2\cos x\right)\,\mathrm{d}x=\dfrac{1}{3}x^3-2\sin x+C.\]
			Để đồ thị hàm số $y=F(x)$ có tâm đối xứng là gốc tọa độ $O$ thì hàm số $F(x)$ phải là hàm số lẻ, nghĩa là
			\allowdisplaybreaks
			\begin{eqnarray*}
				&&F(-x) = -F(x), \forall x \in \mathbb{R}\\
				&\Leftrightarrow& \dfrac{(-x)^3}{3} - 2\sin(-x) + C=-\left(\dfrac{1}{3}x^3-2\sin x+C\right), \forall x \in \mathbb{R} \\
				&\Leftrightarrow& -\dfrac{x^3}{3} + 2\sin x + C=-\dfrac{x^3}{3} + 2\sin x - C, \forall x \in \mathbb{R}\\
				&\Rightarrow&C = -C\\
				&\Rightarrow&2C = 0\\
				&\Rightarrow& C=0.
			\end{eqnarray*}
			Vậy $F(x)=\dfrac{1}{3}x^3-2\sin x$.
			\itemch
			Ta có $f(x) = F'(x) = x^2 - 2\cos x$. Đây là hàm số chẵn.\\
			Xét $f'(x) = 2x + 2\sin x$.\\
			Với $x > 0$ thì $f'(x) > 0$, suy ra hàm số đồng biến trên $(0; +\infty)$.\\
			Mà $f(0) = -2 < 0$ và $f(\pi) = \pi^2 + 2 > 0$ nên phương trình $f(x) = 0$ có đúng $1$ nghiệm dương $x_0 \in (0; \pi)$.\\
			Do hàm số chẵn nên phương trình sẽ có đúng $2$ nghiệm đối nhau là $x_0$ và $-x_0$.\\
			Bảng biến thiên
			\begin{center}
				\begin{tikzpicture}
					\tkzTabInit[nocadre=false,lgt=1.2,espcl=4,deltacl=0.6]
					{$x$ /0.6,$f’(x)$ /0.6,$f(x)$ /2}
					{$-\infty$, $0$, $+\infty$}
					\tkzTabLine{, -, 0, +, }
					\tkzTabVar{+/ $+\infty$, -/ $-2$, +/ $+\infty$}
				\end{tikzpicture}
			\end{center}
			Tại các điểm này $F'(x)$ (tức $f(x)$) đều đổi dấu, vậy đồ thị $y = F(x)$ có hai điểm cực trị nằm về hai phía của trục tung.
		\end{itemchoice}
	}
\end{ex}

	Mô hình toán học sau đây được sử dụng trong quan sát chuyển động của một vật. Trong không gian cho hệ tọa độ $Oxyz$ có $\overrightarrow{i}$, $\overrightarrow{j}$, $\overrightarrow{k}$ lần lượt là các vectơ đơn vị trên các trục $Ox$, $Oy$, $Oz$. Cho hai điểm $A$ và $B$, trong đó điểm $A$ có tọa độ là $(5;5;0)$. Một vật (coi như là một hạt) chuyển động thẳng với tốc độ phụ thuộc thời gian $t(s)$ theo công thức $v(t) = \beta t + 300$ m/s, trong đó $\beta$ là hằng số dương và $0 \le t \le 6$. Ở thời điểm ban đầu $(t=0)$, vật đi qua $A$ với tốc độ $300$ m/s và hướng tới $B$. Sau $2$ giây kể từ thời điểm ban đầu, vật đi được quãng đường $604$ m. Gọi $\overrightarrow{u}=(a;b;c)$ là vectơ cùng hướng với vectơ $\overrightarrow{AB}$. Biết $|\overrightarrow{u}|=1$ và góc giữa vectơ $\overrightarrow{u}$ lần lượt với các vectơ $\overrightarrow{i}, \overrightarrow{j}, \overrightarrow{k}$ có số đo tương ứng bằng $60^\circ$, $60^\circ$, $45^\circ$.
	\choiceTF[t]
	{\True $a = \cos 60^\circ$}
	{Phương trình đường thẳng $AB$ là $\dfrac{x-6}{1} = \dfrac{y-6}{1} = \dfrac{z}{2}$}
	{$\beta = 3$}
	{Giả sử sau $5$ giây kể từ thời điểm ban đầu, vật đến điểm $B(x_B; y_B; z_B)$. Khi đó ta có $x_B = 781$}
	\loigiai{
		\begin{itemchoice}
			\itemch Gọi $\alpha_1$, $\alpha_2$, $\alpha_3$ lần lượt là góc tạo bởi vectơ $\overrightarrow{u}$ và các vectơ đơn vị $\overrightarrow{i}$, $\overrightarrow{j}$, $\overrightarrow{k}$.\\
			Ta có $\overrightarrow{u} = (a; b; c) = (\cos \alpha_1; \cos \alpha_2; \cos \alpha_3)$.\\
			Theo giả thiết $\alpha_1 = 60^\circ$ nên $a = \cos 60^\circ = \dfrac{1}{2}$.
			\itemch Từ kết quả trên, ta có tọa độ vectơ chỉ phương của đường thẳng $AB$ là
			\[\overrightarrow{u} = (\cos 60^\circ; \cos 60^\circ; \cos 45^\circ) = \left(\dfrac{1}{2}; \dfrac{1}{2}; \dfrac{\sqrt{2}}{2}\right).\]
			Chọn vectơ chỉ phương khác cùng phương là $\overrightarrow{u}_{AB} = 2\overrightarrow{u} = (1; 1; \sqrt{2})$.\\
			Phương trình chính tắc của đường thẳng $AB$ đi qua $A(5;5;0)$ và có vectơ chỉ phương $\overrightarrow{u}_{AB}(1; 1; \sqrt{2})$ là
			$$\dfrac{x-5}{1} = \dfrac{y-5}{1} = \dfrac{z}{\sqrt{2}}.$$
			\itemch Ta có quãng đường vật đi được là nguyên hàm của vận tốc
			\[S(t) = \displaystyle\int v(t) \mathrm{d}t = \displaystyle\int (\beta t + 300) \mathrm{d}t = \dfrac{\beta t^2}{2} + 300t + C.\]
			Vì tại thời điểm ban đầu $t=0$, vật bắt đầu đi từ $A$ nên $S(0)=0 \Rightarrow C=0$.\\
			Theo giả thiết, sau $2$ giây vật đi được $604\mathrm{m}$, ta có
			\[S(2) = \dfrac{\beta \cdot 2^2}{2} + 300 \cdot 2 = 604 \Leftrightarrow 2\beta + 600 = 604 \Leftrightarrow 2\beta = 4 \Leftrightarrow \beta = 2.\]
			\itemch Sau $5$ giây, quãng đường vật đi được là
			$S(5) = \dfrac{2 \cdot 5^2}{2} + 300 \cdot 5 = 25 + 1500 = 1525$ m.\\
			Ta có $\overrightarrow{AB}$ cùng hướng với $\overrightarrow{u}$ và có độ dài bằng quãng đường đi được nên
			\[\overrightarrow{AB} = S(5) \cdot \overrightarrow{u} = 1525 \cdot \left(\dfrac{1}{2}; \dfrac{1}{2}; \dfrac{\sqrt{2}}{2}\right) = \left(\dfrac{1525}{2}; \dfrac{1525}{2}; \dfrac{1525\sqrt{2}}{2}\right).\]
			Tọa độ điểm $B(x_B; y_B; z_B)$ thỏa mãn $\overrightarrow{AB} = (x_B - 5; y_B - 5; z_B - 0)$.\\
			Suy ra tọa độ $x_B$
			\[x_B - 5 = \dfrac{1525}{2} \Rightarrow x_B = 5 + 762{,}5 = 767{,}5.\]
		\end{itemchoice}
	}
\end{ex}

\begin{ex}%[2D6V2-4]%[TDMT12]%[Văn Minh Thoại]
	Thống kê cho thấy học tập tốt chăm chỉ làm việc thì tỉ lệ thành công rực rỡ là $90$\%, còn lại thì tỉ lệ thành công rực rỡ là $20$\%. Theo thống kê thì trong dân số chỉ có $15$\% những người thành công rực rỡ là những người học tốt chăm chỉ làm việc. Chọn ngẫu nhiên một người trong dân số. Gọi $A$ là biến cố chọn được người học tập tốt chăm chỉ làm việc, $B$ là biến cố chọn được người thành công rực rỡ.
	\choiceTF[t]
	{\True $\mathrm{P}(B\mid A)=0{,}9$}
	{\True $\mathrm{P}\left(B\mid\overline{A}\right)=0{,}2$}
	{\True $\mathrm{P}(A\mid B)=0{,}15$}
	{$\mathrm{P}(A)=0{,}03$}
	\loigiai{
		\begin{itemchoice}
			\itemch Tỉ lệ thành công rực rỡ của người học tập tốt chăm chỉ làm việc là $90$\%, suy ra $\mathrm{P}(B\mid A)=0{,}9$.
			\itemch Tỉ lệ thành công rực rỡ của những người còn lại là $20$\%, suy ra $\mathrm{P}\left(B\mid\overline{A}\right)=0{,}2$.
			\itemch Trong số những người thành công rực rỡ, có $15$\% là người học tập tốt chăm chỉ làm việc, suy ra $\mathrm{P}(A\mid B)=0{,}15$.
			\itemch Để tính $\mathrm{P}(A)$, ta sử dụng công thức Bayes
			\begin{eqnarray*}
				&&\mathrm{P}(A\mid B)=\dfrac{\mathrm{P}(A)\cdot \mathrm{P}(B\mid A)}{\mathrm{P}(A)\cdot \mathrm{P}(B\mid A)+\mathrm{P}\left(\overline{A}\right) \cdot\mathrm{P}\left(B\mid\overline{A}\right)}\\
				&\Leftrightarrow&0{,}15= \dfrac{\mathrm{P}(A) \cdot 0{,}9}{\mathrm{P}(A) \cdot 0{,}9+(1-\mathrm{P}(A))\cdot 0{,}2} \\
				&\Leftrightarrow& \mathrm{P}(A)\approx 0{,}0377.
			\end{eqnarray*}
		\end{itemchoice}
	}
\end{ex}

\begin{ex}%[2D4C3-2]%[TDMT12]%[Võ Hiệp]
	\immini[thm]{
		Cho hình chữ nhật $ABCD$ có $AB = 40\mathrm{\,cm}$, $BC = 30\mathrm{\,cm}$. Cho một chú bọ $X$ ban đầu đang đứng ở đỉnh $A$ và được coi như một điểm nhỏ. Chú bọ $X$ bò theo đường thẳng đến điểm gần nhất $M$ thỏa mãn khoảng cách đến $C$ bằng khoảng cách đến $AB$. 	\par
	\shortans[0]{605}
	\loigiai{
		\immini{
			Đặt hệ trục tọa độ $Oxy$ với đơn vị trên trục bằng $1\mathrm{\,(cm)}$ sao cho gốc $O$ trùng với đỉnh $A$, $B\in Ox$, $D\in Oy$.\\

			Gọi $M(x; y)$. Suy ra
			$MC = \sqrt{(x-40)^2 + (y-30)^2}$, khoảng cách từ $M$ đến cạnh $AB$ là $\mathrm{d}(M,AB) = |y|$.

		\noindent
		Theo đề bài ta có
		\allowdisplaybreaks
		\begin{align*}
			MC= \mathrm{d}(M,AB)
			&\Rightarrow \sqrt{(x-40)^2 + (y-30)^2} = |y|\\
			&\Rightarrow (x-40)^2 + (y-30)^2 = y^2\\
			&\Rightarrow (x-40)^2 + y^2-60y + 900 = y^2\\
			&\Rightarrow y=\dfrac{(x-40)^2}{60}+15.
		\end{align*}
		Suy ra điểm $M$ thuộc parabol $(P)\colon y=\dfrac{(x-40)^2}{60}+15$.\\
		Do bọ di chuyển trong hình chữ nhật $ABCD$ nên ta có $x>0$, $y>0$. \\
		Khi bọ đến điểm thuộc cạnh $CD$, ta có $y=30$. Suy ra
		\[\dfrac{(x-40)^2}{60}+15=30\Leftrightarrow (x-40)^2=900\Leftrightarrow \hoac{&x=10>0\\&x=-70<0.}\]
		Khi bọ đến điểm thuộc cạnh $BC$, ta có $x=40$.\\
		Ta có $AM=\sqrt{x^2+y^2}=\sqrt{x^2+\left[\dfrac{(x-40)^2}{60}+15\right]^2}$, với $10\le x\le 40$.\\
		Xét hàm số $f(x)=x^2+\left[\dfrac{(x-40)^2}{60}+15\right]^2$ trên đoạn $[10; 40]$.\\
		Đạo hàm $f'(x)=2x + 2 \cdot \left[ \dfrac{(x-40)^2}{60} + 15 \right] \cdot \left[ \dfrac{2(x-40)}{60} \right]$.\\
		Đặt $u=x-40\Rightarrow x=u+40$.\\
		Khi đó hàm số $f'(x)$ trở thành
		$g(u)=2(u + 40) + 2\left( \dfrac{u^2}{60} + 15 \right) \cdot \dfrac{u}{30}$, $(-30\le u\le 0)$. \\
		Xét
		\allowdisplaybreaks
		\begin{align*}
			g(u)=0
			&\Leftrightarrow
			(u + 40) + \left( \dfrac{u^2}{60} + 15 \right) \cdot \dfrac{u}{30} = 0\\
			&\Leftrightarrow
			u^3 + 2\,700u + 72\,000 = 0\\
			&\Leftrightarrow u\approx -22{,}47.

		Xác suất cần tìm là $P = \mathrm{P}(A|B) = 1 - \mathrm{P}(\overline{A}|B) = 1 - \dfrac{\mathrm{P}(\overline{A} \cap B)}{\mathrm{P}(B)}$.\\
		
		\noindent\textbf{Bước 1: Tính $\mathrm{P}(B)$}\\

			\item $\mathrm{P}(H_0) = \dfrac{\mathrm{C}_5^3}{\mathrm{C}_{12}^3} = \dfrac{1}{22}$. Lúc này hộp II có $5$ bi đỏ, $7$ bi xanh. Xác suất lấy $2$ bi đỏ từ hộp II là $\mathrm{P}(B|H_0) = \dfrac{\mathrm{C}_5^2}{\mathrm{C}_{12}^2} = \dfrac{5}{33}$.
		\end{itemize}
		Theo công thức xác suất toàn phần, ta có:
		\allowdisplaybreaks
		\begin{align*}
			\mathrm{P}(B) &= \mathrm{P}(H_3)\mathrm{P}(B|H_3) + \mathrm{P}(H_2)\mathrm{P}(B|H_2) + \mathrm{P}(H_1)\mathrm{P}(B|H_1) + \mathrm{P}(H_0)\mathrm{P}(B|H_0) \\
			&= \dfrac{7}{44} \cdot \dfrac{14}{33} + \dfrac{21}{44} \cdot \dfrac{7}{22} + \dfrac{14}{44} \cdot \dfrac{5}{22} + \dfrac{1}{22} \cdot \dfrac{5}{33} = \dfrac{289}{968}.
		\end{align*}

		Nếu biết chắc chắn $2$ viên bi lấy ra chính là $2$ viên bi mang từ hộp I sang, thì xác suất để chúng đều có màu đỏ bằng đúng xác suất lấy được $2$ viên bi đỏ trong một phép thử lấy $2$ viên bi từ hộp I ban đầu. Ta có $\mathrm{P}(B|\overline{A}) = \dfrac{\mathrm{C}_7^2}{\mathrm{C}_{12}^2} = \dfrac{21}{66} = \dfrac{7}{22}$.\\
		Suy ra $\mathrm{P}(\overline{A} \cap B) = \mathrm{P}(\overline{A}) \cdot \mathrm{P}(B|\overline{A}) = \dfrac{1}{22} \cdot \dfrac{7}{22} = \dfrac{7}{484} = \dfrac{14}{968}$.\\
		
		\noindent\textbf{Bước 3: Tính giá trị cần tìm}\\
		Xác suất $P = \mathrm{P}(A|B) = 1 - \dfrac{\mathrm{P}(\overline{A} \cap B)}{\mathrm{P}(B)} = 1 - \dfrac{14/968}{289/968} = 1 - \dfrac{14}{289} = \dfrac{275}{289}$.\\
		Vậy $1000P = 1000 \cdot \dfrac{275}{289} \approx 951{,}557 \approx 952$.
	}
\end{ex}
